Run this single SQL migration in your Supabase SQL Editor to create the tables for OTP authentication, user profiles, and scan logs: 

-- 1. Create User Profiles Table
CREATE TABLE IF NOT EXISTS public.profiles (
    id UUID REFERENCES auth.users ON DELETE CASCADE PRIMARY KEY,
    email TEXT NOT NULL,
    tier TEXT DEFAULT 'free',
    scans_remaining INT DEFAULT 20,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 2. Create Scans Table for Plagiarism Audit Reports
CREATE TABLE IF NOT EXISTS public.scans (
    id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
    user_id UUID REFERENCES auth.users ON DELETE SET NULL,
    repo TEXT NOT NULL,
    commit_sha TEXT NOT NULL,
    commit_message TEXT,
    author_name TEXT,
    verdict TEXT NOT NULL,
    plagiarism_score INT NOT NULL,
    plagiarism_category TEXT,
    flagged_snippets JSONB,
    reasoning TEXT,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 3. Enable Row Level Security (RLS)
ALTER TABLE public.profiles ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.scans ENABLE ROW LEVEL SECURITY;

-- 4. Set RLS Policies
CREATE POLICY "Users can view own profile" ON public.profiles FOR SELECT USING (auth.uid() = id);
CREATE POLICY "Anyone can view scans" ON public.scans FOR SELECT USING (true);
CREATE POLICY "Users can insert scans" ON public.scans FOR INSERT WITH CHECK (true);

-- 5. Auto-create Profile Trigger upon Email OTP Sign Up
CREATE OR REPLACE FUNCTION public.handle_new_user()
RETURNS TRIGGER AS $$
BEGIN
  INSERT INTO public.profiles (id, email)
  VALUES (new.id, new.email)
  ON CONFLICT (id) DO NOTHING;
  RETURN NEW;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

CREATE OR REPLACE TRIGGER on_auth_user_created
  AFTER INSERT ON auth.users
  FOR EACH ROW EXECUTE FUNCTION public.handle_new_user();
